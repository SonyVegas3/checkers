	function coin_offer(d) {
		emblem(d, '<img src="ui/images/v2/coin.svg" alt="" />', 'v2m-pulse');

		var content_all = part(d, 'content');
		if(content_all) {
			coin_amount(content_all.firstElementChild || content_all);
		}
		who(d, true);

		// only these games close the offer by itself after 10 seconds
		if(!/^\/(game|checkers|backgammon|domino)\//.test(location.pathname)) return;

		var content = part(d, 'content');
		if(!content || content.querySelector('.v2m-timer')) return;

		var holder = content.firstElementChild || content;
		var timer = document.createElement('div');
		timer.className = 'v2m-timer';
		timer.innerHTML = '<i></i>';
		holder.appendChild(timer);

		// "closes automatically" on the left, "10 s" on the right
		var note = document.createElement('div');
		note.className = 'v2m-note';
		var texts = t('auto_close').split(' · ');
		for(var i = 0; i < texts.length; i++) {
			var span = document.createElement('span');
			span.textContent = texts[i];
			note.appendChild(span);
		}
		holder.appendChild(note);
	}

	// "... გეთამაშოთ <b>500</b> <img> მონეტაზე" -> a big amount, then the sentence without it (georgian only)
	function coin_amount(holder) {
		var line = holder ? holder.firstElementChild : null;
		if(lang_id() != 'geo' || !line || holder.querySelector('.v2m-amount')) return;

		var b = line.querySelector('b');
		var img = line.querySelector('img[src*="images/coin.png"]');
		var m = text_of(line).match(/^მოწინააღმდეგეს სურს გეთამაშოთ (\d+) მონეტაზე$/);
		if(!b || !img || !m || text_of(b) != m[1]) return;

		var amount = document.createElement('div');
		amount.className = 'v2m-amount';
		amount.innerHTML = '<img src="ui/images/v2/coin.svg" alt="" /><b></b>';
