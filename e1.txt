    this.rematch_watch = function(args) {
        var opponent_id = String(args.opponent_id || '');

        if(
            !opponent_id
                ||
            obj.self.rematch_opponent
        ) {
            return;
        }

        obj.self.rematch_opponent = opponent_id;

        if(!obj.self.checker_interval) {
            obj.self.update = -1;
            obj.self.checker({
                'action': 'start',
                'interval': 2000
            });
        }
    }
