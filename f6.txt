	function emblem(d, html, mods, plain) {
		var inner = part(d, 'inner');
		if(!inner || (inner.firstElementChild && has(inner.firstElementChild, 'v2m-emb'))) return;

		if(!plain) {
			d.classList.add('v2m-e');
		}

		var emb = document.createElement('div');
		emb.className = 'v2m-emb'+(mods ? ' '+mods : '');
		emb.innerHTML = html;
		inner.insertBefore(emb, inner.firstChild);
	}
